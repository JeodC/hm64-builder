#### BUILD 2SHIP ####
git clone https://github.com/HarbourMasters/2ship2harkinian.git
cd 2ship2harkinian

git fetch --tags
git checkout $(git describe --tags `git rev-list --tags --max-count=1`)

# Clone the submodules
git submodule update --init

# Patch for old git
cd libultraship
patch -p1 < ../../soh2.patch
cd ..

mkdir build-soh && cd build-soh

export CFLAGS="-O3 -Wall -Wextra -Wno-return-type -Wno-unused-parameter -Wno-unused-function -Wno-unused-variable -Wno-macro-redefined -Wno-unknown-warning-option"
export CXXFLAGS="-O3 -Wall -Wextra -Wno-return-type -Wno-unused-parameter -Wno-unused-function -Wno-unused-variable -Wno-macro-redefined -Wno-unknown-warning-option"

# Verify where clang is or use the default 'clang' which is v18 on 24.04
if ! command -v clang-18 &> /dev/null; then
    export CC=clang
    export CXX=clang++
else
    export CC=clang-18
    export CXX=clang++-18
fi

cmake .. -GNinja \
-DUSE_OPENGLES=1 \
-DBUILD_CROWD_CONTROL=0 \
-DCMAKE_BUILD_TYPE=Release

cmake --build . --target Generate2ShipOtr -j$(nproc)
cmake --build . -j$(nproc)

# Stage shipped libraries. Filenames MUST match the binary's DT_NEEDED entries
# exactly (verify with `readelf -d 2s2h.elf | grep NEEDED`). We use `cp -L` to
# follow symlinks so each destination is a real file with the right name;
# wildcards + `find -delete` can produce broken symlinks that silently fall
# through to the system library. libSDL2 is NOT shipped (device provides it).
# 2s2h adds vorbis/spdlog to SoH's set.
mkdir libs
cp -L /usr/lib/aarch64-linux-gnu/libz.so.1 libs/libz.so.1
cp -L /usr/lib/aarch64-linux-gnu/libogg.so.0 libs/libogg.so.0
cp -L /usr/lib/aarch64-linux-gnu/libpng16.so.16 libs/libpng16.so.16
cp -L /usr/lib/aarch64-linux-gnu/libvorbis.so.0 libs/libvorbis.so.0
cp -L /usr/lib/aarch64-linux-gnu/libvorbisenc.so.2 libs/libvorbisenc.so.2
cp -L /usr/lib/aarch64-linux-gnu/libvorbisfile.so.3 libs/libvorbisfile.so.3
cp -L /usr/local/lib/libspdlog.so.1 libs/libspdlog.so.1
cp -L /usr/local/lib/libbz2.so.1 libs/libbz2.so.1
cp -L /usr/local/lib/libzip.so.5 libs/libzip.so.5
cp -L /usr/local/lib/libtinyxml2.so.10 libs/libtinyxml2.so.10
cp -L /usr/local/lib/libopus.so.0 libs/libopus.so.0
cp -L /usr/local/lib/libopusfile.so.0 libs/libopusfile.so.0

# Verify: every staged lib is in NEEDED, every expected NEEDED lib is staged.
needed=$(readelf -d mm/2s2h.elf | awk -F'[][]' '/NEEDED/ {print $2}')
echo "Binary NEEDED:"; echo "$needed"
echo "Staged:"; ls libs/
for lib in libz.so.1 libogg.so.0 libpng16.so.16 libvorbis.so.0 \
           libvorbisenc.so.2 libvorbisfile.so.3 libspdlog.so.1 \
           libbz2.so.1 libzip.so.5 libtinyxml2.so.10 libopus.so.0 \
           libopusfile.so.0; do
    if ! echo "$needed" | grep -qx "$lib"; then
        echo "ERROR: staged $lib but binary does not NEED it"; exit 1
    fi
    if [ ! -f "libs/$lib" ]; then
        echo "ERROR: $lib missing from libs/"; exit 1
    fi
done

cd ../..
