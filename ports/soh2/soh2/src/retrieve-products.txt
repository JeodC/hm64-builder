#!/bin/bash
set -e  # Exit immediately if a command fails

SRCDIR="$1"
DESTDIR="$2"

# Copy binaries
for elf_file in "$SRCDIR/2ship2harkinian/build-soh/mm/"*.elf; do
    cp "$elf_file" "$DESTDIR/"
done
cp "$SRCDIR/2ship2harkinian/build-soh/mm/2ship.o2r" "$DESTDIR/"

# Replace libs folder
rm -rf "$DESTDIR/libs"
cp -r "$SRCDIR/2ship2harkinian/build-soh/libs" "$DESTDIR/"

# Prepare temporary folder for zipping
TEMP_ZIP_DIR="$DESTDIR/assets/tmp_zip_dir"
rm -rf "$TEMP_ZIP_DIR"
mkdir -p "$TEMP_ZIP_DIR"

# Copy extractor contents into temp folder
mkdir -p "$TEMP_ZIP_DIR/extractor"
cp "$SRCDIR/2ship2harkinian/build-soh/ZAPD/ZAPD.out" "$TEMP_ZIP_DIR/extractor/ZAPD.out"
cp -r "$SRCDIR/2ship2harkinian/mm/assets/extractor/"* "$TEMP_ZIP_DIR/extractor/"

# Copy xml folder into temp folder
cp -r "$SRCDIR/2ship2harkinian/mm/assets/xml" "$TEMP_ZIP_DIR/"

# Create the zip from temp folder
(
  cd "$TEMP_ZIP_DIR"
  zip -r "$DESTDIR/assets/extractor.zip" *
)

# Clean up
rm -rf "$TEMP_ZIP_DIR"
