#!/bin/bash
set -e  # Exit immediately if a command fails

SRCDIR="$1"
DESTDIR="$2"

# Copy binaries
for elf_file in "$SRCDIR/2ship2harkinian/build-soh/mm/"*.elf; do
    cp "$elf_file" "$DESTDIR/"
done
cp "$SRCDIR/2ship2harkinian/build-soh/mm/2ship.o2r" "$DESTDIR/"

# Replace libs folder
rm -rf "$DESTDIR/libs"
cp -r "$SRCDIR/2ship2harkinian/build-soh/libs" "$DESTDIR/"

# Recreate extractor.zip
EXTRACTOR_DIR="$DESTDIR/assets/extractor"
rm -rf "$EXTRACTOR_DIR"
mkdir -p "$EXTRACTOR_DIR"

# Copy ZAPD.out
cp "$SRCDIR/2ship2harkinian/build-soh/ZAPD/ZAPD.out" "$EXTRACTOR_DIR/ZAPD.out"

# Copy extractor contents
cp -r "$SRCDIR/2ship2harkinian/mm/assets/extractor/"* "$EXTRACTOR_DIR"

# Copy xml folder **inside the assets folder temporarily for zipping**
TEMP_ZIP_DIR="$DESTDIR/assets/tmp_zip_dir"
mkdir -p "$TEMP_ZIP_DIR"
cp -r "$DESTDIR/assets/extractor" "$TEMP_ZIP_DIR/"
cp -r "$SRCDIR/2ship2harkinian/mm/assets/xml" "$TEMP_ZIP_DIR/"

(
  cd "$TEMP_ZIP_DIR"
  zip -r "$DESTDIR/assets/extractor.zip" *
)

rm -rf "$TEMP_ZIP_DIR"
