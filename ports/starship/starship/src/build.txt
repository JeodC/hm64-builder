# against a modern SDL2 at build time. We do NOT ship libSDL2-2.0.so.0 —

cd SDL

cmake ..
make -j$(nproc)
make install
cd ../..

#### Starship ####

git clone --recursive https://github.com/HarbourMasters/starship.git
cd starship

git fetch --tags
if [[ "${FORCE_HEAD:-false}" == "true" ]]; then
  echo "FORCE_HEAD=true — building from branch HEAD (nightly)"
else
  git checkout $(git describe --tags `git rev-list --tags --max-count=1`)
fi

# Clone the submodules
git submodule update --init

# Verify where clang is or use the default 'clang' which is v18 on 24.04
if ! command -v clang-18 &> /dev/null; then
    export CC=clang
    export CXX=clang++
else
    export CC=clang-18
    export CXX=clang++-18
fi

mkdir build-starship && cd build-starship
cmake .. -GNinja \
-DUSE_OPENGLES=1 \
-DCMAKE_BUILD_TYPE=Release \
-DCMAKE_CXX_FLAGS=-ldl

cmake --build . -j$(nproc) --config Release --target GeneratePortO2R
cmake --build . -j$(nproc)

# Stage shipped libraries. Filenames MUST match the binary's DT_NEEDED entries
# exactly (verify with `readelf -d Starship | grep NEEDED`). We use `cp -L` to
# follow symlinks so each destination is a real file with the right name;
# wildcards + `find -delete` can produce broken symlinks that silently fall
# through to the system library. libSDL2 is NOT shipped (device provides it).
# tinyxml2/spdlog/nlohmann-json/GSL are static-linked, not runtime deps.
# vorbis/ogg are dynamic deps Starship inherits via libSDL2_mixer/etc., shipped
# defensively for handhelds that don't have them in /usr/lib.
mkdir libs
cp -L /usr/lib/aarch64-linux-gnu/libz.so.1         ./libs/libz.so.1
cp -L /usr/lib/aarch64-linux-gnu/libogg.so.0       ./libs/libogg.so.0
cp -L /usr/lib/aarch64-linux-gnu/libvorbis.so.0    ./libs/libvorbis.so.0
cp -L /usr/lib/aarch64-linux-gnu/libvorbisenc.so.2 ./libs/libvorbisenc.so.2
cp -L /usr/lib/aarch64-linux-gnu/libvorbisfile.so.3 ./libs/libvorbisfile.so.3
cp -L /usr/local/lib/libzip.so.5                   ./libs/libzip.so.5

# Verify: every staged lib is in NEEDED, every expected NEEDED lib is staged.
needed=$(readelf -d Starship | awk -F'[][]' '/NEEDED/ {print $2}')
echo "Binary NEEDED:"; echo "$needed"
echo "Staged:"; ls libs/
for lib in libz.so.1 libogg.so.0 libvorbis.so.0 libvorbisenc.so.2 libvorbisfile.so.3 libzip.so.5; do
    if ! echo "$needed" | grep -qx "$lib"; then
        echo "ERROR: staged $lib but binary does not NEED it"; exit 1
    fi
    if [ ! -f "libs/$lib" ]; then
        echo "ERROR: $lib missing from libs/"; exit 1
    fi
done

cd ../..
