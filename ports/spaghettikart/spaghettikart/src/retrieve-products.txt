#!/bin/bash

SRCDIR=$1
DESTDIR=$2

cd "$DESTDIR" || { echo "DESTDIR not found: $DESTDIR"; exit 1; }

cp "$SRCDIR/SpaghettiKart/build-spaghettikart/Spaghettify" "$DESTDIR/"
cp "$SRCDIR/SpaghettiKart/build-spaghettikart/Spaghettify.pdb" "$DESTDIR/"
cp "$SRCDIR/SpaghettiKart/build-spaghettikart/spaghetti.o2r" "$DESTDIR/"
wget -O "$DESTDIR/gamecontrollerdb.txt" "https://raw.githubusercontent.com/mdqinc/SDL_GameControllerDB/master/gamecontrollerdb.txt"

rm -rf ./libs
cp -r "$SRCDIR/SpaghettiKart/build-spaghettikart/libs" .

mkdir -p "$DESTDIR/tools"

# Copy torch binary
cp "$SRCDIR/SpaghettiKart/build-spaghettikart/TorchExternal/src/TorchExternal-build/torch" \
   "$DESTDIR/tools/torch"

# Remove old tools content
rm -rf "$DESTDIR/tools/yamls"
rm -f "$DESTDIR/tools/config.yml"

# Copy new tools content
cp "$SRCDIR/SpaghettiKart/config.yml" "$DESTDIR/tools"

# Zip the entire folder as "assets/" inside the zip
pushd "$SRCDIR/SpaghettiKart"
zip -r "$DESTDIR/tools/assets.zip" yamls/ meta/ include/
popd > /dev/null

echo "Packaging complete. You will also need to update otrgen by hand."
