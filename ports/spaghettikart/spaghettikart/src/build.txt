# ========================
# System setup
# ========================

apt update
apt install -y \
  build-essential \
  git \
  python3 \
  python3-pip \
  wget \
  gnupg \
  lsb-release \
  pkg-config \
  unzip \
  cmake

# ========================
# LLVM / Clang-18
# ========================

wget https://apt.llvm.org/llvm.sh
chmod +x llvm.sh
./llvm.sh 18

apt install -y \
  clang-18 \
  clang-tools-18 \
  libc++-18-dev \
  libc++abi-18-dev

# ========================
# Ninja
# ========================

git clone https://github.com/ninja-build/ninja.git
cd ninja
git checkout v1.11.1
./configure.py --bootstrap
install -m 755 ninja /usr/local/bin/ninja
cd ..

# ========================
# Global toolchain
# ========================

export CC=clang-18
export CXX=clang++-18
export CFLAGS="-O3 -DNDEBUG -flto"
export CXXFLAGS="-O3 -DNDEBUG -flto -stdlib=libc++"
export LDFLAGS="-stdlib=libc++ -ldl -lrt -lm -pthread"
export PKG_CONFIG_PATH="/usr/local/lib/pkgconfig:/usr/lib/pkgconfig"

# ========================
# SpaghettiKart
# ========================

git clone https://github.com/HarbourMasters/SpaghettiKart.git
cd SpaghettiKart
git submodule update --init --recursive

cmake -GNinja -B build-spaghettikart \
  -DCMAKE_BUILD_TYPE=Release \
  -DCMAKE_C_COMPILER=clang-18 \
  -DCMAKE_CXX_COMPILER=clang++-18 \
  -DCMAKE_CXX_STANDARD=20 \
  -DCMAKE_CXX_FLAGS_RELEASE="-O3 -DNDEBUG -flto -stdlib=libc++" \
  -DCMAKE_EXE_LINKER_FLAGS="-stdlib=libc++ -ldl -lrt -lm -pthread" \
  -DUSE_OPENGLES=1 \
  -DBUILD_CROWD_CONTROL=0

ninja -C build-spaghettikart GenerateO2R
ninja -C build-spaghettikart -j$(nproc)

# ========================
# Gather runtime libs
# ========================

mkdir libs
cp /usr/lib/aarch64-linux-gnu/libogg.so.[0-9]* ./libs/
cp /usr/lib/aarch64-linux-gnu/libvorbis.so.[0-9]* ./libs/
cp /usr/lib/aarch64-linux-gnu/libvorbisenc.so.[0-9]* ./libs/
cp /usr/lib/aarch64-linux-gnu/libvorbisfile.so.[0-9]* ./libs/
cp /usr/lib/aarch64-linux-gnu/libz.so.[0-9]* ./libs/
cp /usr/local/lib/libzip.so.[0-9]* ./libs/
cp /usr/local/lib/libtinyxml2.so.[0-9]* ./libs/
find ./libs/ -name "*.so.[0-9]*.[0-9]*" -delete

cd ../..
