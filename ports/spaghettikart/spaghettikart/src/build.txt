## SpaghettiKart ##

git clone "${REPO_URL:-https://github.com/HarbourMasters/SpaghettiKart.git}" SpaghettiKart
cd SpaghettiKart

# Checkout latest tag, or branch HEAD if FORCE_HEAD is set
git fetch --tags
if [[ -n "${REF}" ]]; then
  echo "REF=$REF — building from explicit ref"
  git checkout "$REF"
elif [[ "${FORCE_HEAD:-false}" == "true" ]]; then
  echo "FORCE_HEAD=true — building from branch HEAD (nightly)"
else
  git checkout $(git describe --tags `git rev-list --tags --max-count=1`)
fi

# Initialize and sync submodules
git submodule sync
git submodule update --init --recursive

## Build
mkdir build-spaghettikart && cd build-spaghettikart
cmake .. -GNinja \
  -DCMAKE_C_COMPILER=clang-18 \
  -DCMAKE_CXX_COMPILER=clang++-18 \
  -DUSE_OPENGLES=1 \
  -DCMAKE_BUILD_TYPE=Debug \
  -DCMAKE_C_FLAGS="-fno-strict-aliasing -funsigned-char" \
  -DCMAKE_CXX_FLAGS="-fno-strict-aliasing -funsigned-char" \
  -DCMAKE_EXE_LINKER_FLAGS="-ldl -pthread -Wl,--no-relax"

cmake --build . -j$(nproc) --config Debug --target GenerateO2R
cmake --build . -j$(nproc)

# Stage shipped libraries. Filenames MUST match the binary's DT_NEEDED entries
# exactly (verify with `readelf -d Spaghettify | grep NEEDED`). We use `cp -L`
# to follow symlinks so each destination is a real file with the right name;
# wildcards + `find -delete` can produce broken symlinks that silently fall
# through to the system library. libSDL2 is NOT shipped (device provides it).
# We also don't ship the audio-driver libs (libasound, libjack, libpulse)
# that appear in NEEDED from SDL2's audio subsystem link — devices provide
# those. Binary doesn't NEED libbz2, so we don't ship it either.
mkdir libs
cp -L /usr/lib/aarch64-linux-gnu/libogg.so.0 ./libs/libogg.so.0
cp -L /usr/lib/aarch64-linux-gnu/libvorbis.so.0 ./libs/libvorbis.so.0
cp -L /usr/lib/aarch64-linux-gnu/libvorbisenc.so.2 ./libs/libvorbisenc.so.2
cp -L /usr/lib/aarch64-linux-gnu/libvorbisfile.so.3 ./libs/libvorbisfile.so.3
cp -L /usr/lib/aarch64-linux-gnu/libz.so.1 ./libs/libz.so.1
cp -L /usr/local/lib/libzip.so.5 ./libs/libzip.so.5
cp -L /usr/local/lib/libtinyxml2.so.10 ./libs/libtinyxml2.so.10

# Verify: every staged lib is in NEEDED, every expected NEEDED lib is staged.
needed=$(readelf -d Spaghettify | awk -F'[][]' '/NEEDED/ {print $2}')
echo "Binary NEEDED:"; echo "$needed"
echo "Staged:"; ls libs/
for lib in libogg.so.0 libvorbis.so.0 libvorbisenc.so.2 libvorbisfile.so.3 \
           libz.so.1 libzip.so.5 libtinyxml2.so.10; do
    if ! echo "$needed" | grep -qx "$lib"; then
        echo "ERROR: staged $lib but binary does not NEED it"; exit 1
    fi
    if [ ! -f "libs/$lib" ]; then
        echo "ERROR: $lib missing from libs/"; exit 1
    fi
done

cd ../..
