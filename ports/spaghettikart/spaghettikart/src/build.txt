## SpaghettiKart ##

git clone https://github.com/HarbourMasters/SpaghettiKart.git
cd SpaghettiKart

# Checkout latest tag
git fetch --tags
git checkout $(git describe --tags `git rev-list --tags --max-count=1`)


# Grab SpaghettiKart PR
git remote add jeodc https://github.com/JeodC/SpaghettiKart.git 2>/dev/null || true
git fetch jeodc pull/650/head:pr-650-spaghettikart
git checkout pr-650-spaghettikart

# Initialize and sync submodules
git submodule sync
git submodule update --init --recursive

# Grab libultraship PR
cd libultraship
git remote add jeodc-lib https://github.com/JeodC/libultraship.git 2>/dev/null || true
git fetch jeodc-lib pull/1004/head:pr-1004-libultraship
git checkout pr-1004-libultraship
cd ..

## Build
mkdir build-spaghettikart && cd build-spaghettikart
cmake .. -GNinja \
  -DCMAKE_C_COMPILER=clang-18 \
  -DCMAKE_CXX_COMPILER=clang++-18 \
  -DUSE_OPENGLES=1 \
  -DCMAKE_BUILD_TYPE=Debug \
  -DCMAKE_C_FLAGS="-fno-strict-aliasing -funsigned-char" \
  -DCMAKE_CXX_FLAGS="-fno-strict-aliasing -funsigned-char" \
  -DCMAKE_EXE_LINKER_FLAGS="-ldl -pthread -Wl,--no-relax"

cmake --build . -j$(nproc) --config Debug --target GenerateO2R
cmake --build . -j$(nproc)

# Collect system libraries
mkdir libs
cp /usr/lib/aarch64-linux-gnu/libogg.so.[0-9]* ./libs/
cp /usr/lib/aarch64-linux-gnu/libvorbis.so.[0-9]* ./libs/
cp /usr/lib/aarch64-linux-gnu/libvorbisenc.so.[0-9]* ./libs/
cp /usr/lib/aarch64-linux-gnu/libvorbisfile.so.[0-9]* ./libs/
cp /usr/lib/aarch64-linux-gnu/libz.so.[0-9]* ./libs/

# Libraries built from source
cp /usr/local/lib/libbz2.so.[0-9]* libs/
cp /usr/local/lib/libzip.so.[0-9]* ./libs/
cp /usr/local/lib/libtinyxml2.so.[0-9]* ./libs/

find ./libs/ -name "*.so.[0-9]*.[0-9]*" -delete

cd ../..
