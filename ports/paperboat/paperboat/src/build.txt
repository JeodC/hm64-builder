#### Paperboat ####

git clone --recursive "${REPO_URL:-https://github.com/HarbourMasters/paperboat.git}" paperboat
cd paperboat

git fetch --tags
if [[ -n "${REF}" ]]; then
  echo "REF=$REF — building from explicit ref"
  git checkout "$REF"
elif [[ "${FORCE_HEAD:-false}" == "true" ]]; then
  echo "FORCE_HEAD=true — building from branch HEAD (nightly)"
else
  git checkout $(git describe --tags `git rev-list --tags --max-count=1`)
fi

# Clone the submodules
git submodule update --init

# Verify where clang is or use the default 'clang' which is v18 on 24.04
if ! command -v clang-18 &> /dev/null; then
    export CC=clang
    export CXX=clang++
else
    export CC=clang-18
    export CXX=clang++-18
fi

mkdir build-paperboat && cd build-paperboat
cmake .. -GNinja \
-DUSE_OPENGLES=1 \
-DCMAKE_BUILD_TYPE=Release \
-DCMAKE_CXX_FLAGS=-ldl

cmake --build . -j$(nproc) --config Release --target GeneratePortO2R
cmake --build . -j$(nproc)

# Stage shipped libraries. Filenames MUST match the binary's DT_NEEDED entries
# exactly (verify with `readelf -d Paperboat | grep NEEDED`). We use `cp -L`
# to follow symlinks so each destination is a real file with the right name;
# wildcards + `find -delete` can produce broken symlinks that silently fall
# through to the system library. libSDL2 is NOT shipped (device provides it).
# The initial set mirrors sm64-ghostship (same Harbourmasters engine family);
# the verification step will fail the build and reveal the actual NEEDED if
# Paperboat needs more than these two.
mkdir libs
cp -L /usr/local/lib/libzip.so.5 ./libs/libzip.so.5
cp -L /usr/local/lib/libtinyxml2.so.10 ./libs/libtinyxml2.so.10

# Verify: every staged lib is in NEEDED, every expected NEEDED lib is staged.
needed=$(readelf -d Paperboat | awk -F'[][]' '/NEEDED/ {print $2}')
echo "Binary NEEDED:"; echo "$needed"
echo "Staged:"; ls libs/
for lib in libzip.so.5 libtinyxml2.so.10; do
    if ! echo "$needed" | grep -qx "$lib"; then
        echo "ERROR: staged $lib but binary does not NEED it"; exit 1
    fi
    if [ ! -f "libs/$lib" ]; then
        echo "ERROR: $lib missing from libs/"; exit 1
    fi
done

cd ../..
