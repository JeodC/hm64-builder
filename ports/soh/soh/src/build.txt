#### SoH ####
git clone https://github.com/HarbourMasters/Shipwright.git
cd Shipwright

git fetch --tags
#git checkout $(git describe --tags `git rev-list --tags --max-count=1`)

git submodule update --init

mkdir build-soh && cd build-soh

# Verify where clang is or use the default 'clang' which is v18 on 24.04
if ! command -v clang-18 &> /dev/null; then
    export CC=clang
    export CXX=clang++
else
    export CC=clang-18
    export CXX=clang++-18
fi

cmake .. -GNinja \
  -DCMAKE_PREFIX_PATH=/usr/local \
  -DUSE_OPENGLES=1 \
  -DBUILD_REMOTE_CONTROL=1 \
  -DCMAKE_BUILD_TYPE=Release

cmake --build . --target GenerateSohOtr -j$(nproc)
cmake --build . -j$(nproc)

mkdir libs
cp /usr/lib/aarch64-linux-gnu/libz.so.1 libs/
cp /usr/lib/aarch64-linux-gnu/libogg.so.[0-9]* libs/
cp /usr/lib/aarch64-linux-gnu/libpng16.so.16 libs/
cp /usr/local/lib/libbz2.so.[0-9]* libs/
cp /usr/local/lib/libzip.so.[0-9]* libs/
cp /usr/local/lib/libtinyxml2.so.[0-9]* libs/
cp /usr/local/lib/libopus.so.[0-9]* libs/
cp /usr/local/lib/libopusfile.so.[0-9]* libs/
cp /usr/local/lib/libSDL2.so* libs/
cp /usr/local/lib/libSDL2_net.so* libs/

find ./libs/ -name "*.so.[0-9]*.[0-9]*" -delete

cd ../..
