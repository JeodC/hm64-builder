#### BUILD LIBS ####
# Build SDL2 2.32 from source so the other libraries we build below (libzip,
# tinyxml2, opus, etc.) link against a modern SDL2 at build time. We do NOT
# ship libSDL2-2.0.so.0 — every supported device already provides one, often
# with device-specific patches (GL backend, touchscreen, audio). Shipping
# ours via LD_LIBRARY_PATH would override those patches and could regress
# things that currently work.
git clone https://github.com/libsdl-org/SDL.git
cd SDL
git checkout release-2.32.0
mkdir -p build-soh && cd build-soh
cmake .. \
  -DCMAKE_INSTALL_PREFIX=/usr/local \
  -DBUILD_SHARED_LIBS=ON
make -j8
make install
cd ../..

# Build SDL2_net for anchor support. Unlike SDL2, devices DO NOT reliably
# provide libSDL2_net-2.0.so.0, so we ship this one. It links against SDL2
# at runtime but only uses stable SDL2 APIs that work with any 2.x, so it's
# safe to pair our SDL2_net with the device's SDL2.

#### SoH ####
git clone https://github.com/HarbourMasters/Shipwright.git
cd Shipwright

git fetch --tags
git checkout $(git describe --tags `git rev-list --tags --max-count=1`)

git submodule update --init

mkdir build-soh && cd build-soh

# Verify where clang is or use the default 'clang' which is v18 on 24.04
if ! command -v clang-18 &> /dev/null; then
    export CC=clang
    export CXX=clang++
else
    export CC=clang-18
    export CXX=clang++-18
fi

cmake .. -GNinja \
  -DCMAKE_PREFIX_PATH=/usr/local \
  -DUSE_OPENGLES=1 \
  -DBUILD_REMOTE_CONTROL=1 \
  -DCMAKE_BUILD_TYPE=Release

cmake --build . --target GenerateSohOtr -j$(nproc)
cmake --build . -j$(nproc)

# Stage shipped libraries. Filenames MUST match the binary's DT_NEEDED entries
# exactly (verify with `readelf -d soh.elf | grep NEEDED`). We use `cp -L` to
# follow symlinks so each destination is a real file with the right name —
# otherwise `cp` may preserve a symlink pointing to a file not in the zip,
# producing a broken link, and the dynamic loader silently falls through to
# the system library. No wildcards: multiple matching files (SONAME symlink +
# versioned real file) lead to duplicates or deletions by the stale find hack.
# libSDL2 is NOT shipped (device provides it). libSDL2_net IS shipped because
# devices don't reliably provide it and SoH needs it for anchor support.
mkdir libs
cp -L /usr/lib/aarch64-linux-gnu/libz.so.1 libs/libz.so.1
cp -L /usr/lib/aarch64-linux-gnu/libogg.so.0 libs/libogg.so.0
cp -L /usr/lib/aarch64-linux-gnu/libpng16.so.16 libs/libpng16.so.16
cp -L /usr/local/lib/libbz2.so.1 libs/libbz2.so.1
cp -L /usr/local/lib/libzip.so.5 libs/libzip.so.5
cp -L /usr/local/lib/libtinyxml2.so.10 libs/libtinyxml2.so.10
cp -L /usr/local/lib/libopus.so.0 libs/libopus.so.0
cp -L /usr/local/lib/libopusfile.so.0 libs/libopusfile.so.0
cp -L /usr/local/lib/libSDL2_net-2.0.so.0 libs/libSDL2_net-2.0.so.0

# Verify: every staged lib is in the binary's NEEDED list, and every NEEDED
# lib we intend to ship is staged. System libs (libc, libGLESv2, libSDL2,
# etc.) are assumed present on the device and are NOT shipped.
needed=$(readelf -d soh/soh.elf | awk -F'[][]' '/NEEDED/ {print $2}')
echo "Binary NEEDED:"; echo "$needed"
echo "Staged:"; ls libs/
for lib in libz.so.1 libogg.so.0 libpng16.so.16 libbz2.so.1 libzip.so.5 \
           libtinyxml2.so.10 libopus.so.0 libopusfile.so.0 \
           libSDL2_net-2.0.so.0; do
    if ! echo "$needed" | grep -qx "$lib"; then
        echo "ERROR: staged $lib but binary does not NEED it"; exit 1
    fi
    if [ ! -f "libs/$lib" ]; then
        echo "ERROR: $lib missing from libs/"; exit 1
    fi
done

cd ../..
